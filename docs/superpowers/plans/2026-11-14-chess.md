```js
// scripts/fetch-stockfish.js
// Кладёт Stockfish в public/chess/engine. Движок — GPL-3.0: он едет отдельными файлами и работает в воркере
// (отдельная программа рядом с сайтом), поэтому в зависимостях и в бандле его нет. Сборка сверяется по sha256:
// незаметно сменившийся файл — это чужой код в браузере ученика.
const fs = require('node:fs')
const os = require('node:os')
const path = require('node:path')
const crypto = require('node:crypto')
const { execFileSync } = require('node:child_process')

const VERSION = '19.0.1'
const FILES = {
  'stockfish-19-lite-single.js': 'ccb130d7ff29408531c47d859a49f91d59d4f482d05a53eb1a3231a4f624c91a',
  'stockfish-19-lite-single.wasm': 'ee4f115bdb2d32cadf96cada104a05b5ec257d7ad6f57582190b49bc71f096de',
}
const OUT = path.join(__dirname, '..', 'public', 'chess', 'engine')
const sha = (file) => crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')

const tmp = fs.mkdtempSync(path.join(os.tmpdir(), 'jts-stockfish-'))
// shell: на Windows npm — это npm.cmd
execFileSync('npm', ['pack', `stockfish@${VERSION}`, '--silent'], { cwd: tmp, stdio: 'inherit', shell: true })
execFileSync('tar', ['-xzf', `stockfish-${VERSION}.tgz`], { cwd: tmp, stdio: 'inherit' })
fs.mkdirSync(OUT, { recursive: true })
for (const [name, want] of Object.entries(FILES)) {
  const src = path.join(tmp, 'package', 'bin', name)
  const got = sha(src)
  if (got !== want) throw new Error(`${name}: sha256 ${got}, ждали ${want}`)
  fs.copyFileSync(src, path.join(OUT, name))
}
fs.copyFileSync(path.join(tmp, 'package', 'Copying.txt'), path.join(OUT, 'COPYING.txt'))
fs.writeFileSync(path.join(OUT, 'SOURCE.txt'), [
  `Stockfish.js ${VERSION} (Stockfish 19, lite single-threaded WASM build)`,
  'License: GNU General Public License v3.0 — see COPYING.txt in this folder.',
  `Source code: https://github.com/nmrugg/stockfish.js/tree/v${VERSION}`,
  'Upstream engine: https://github.com/official-stockfish/Stockfish',
  'These files are served unmodified and run as a separate program in a Web Worker.',
  '',
].join('\n'))
fs.rmSync(tmp, { recursive: true, force: true })
console.log(`Stockfish ${VERSION} → ${OUT}`)
```

```js
// src/practice/chess/rating.js
// Рейтинг ученика по формуле Эло. Считает браузер: для партий с ботом подделка никому не вредит,
// а к онлайну (часть 4) расчёт переедет на сервер.
export const START_RATING = 800
export const K = 32
export const FLOOR = 100

export const expected = (r, opp) => 1 / (1 + 10 ** ((opp - r) / 400))

export function nextRating(r, opp, score) {
  return Math.max(FLOOR, Math.round(r + K * (score - expected(r, opp))))
}

export function scoreFor(result, color) {
  if (result === '1/2-1/2') return 0.5
  return (result === '1-0') === (color === 'w') ? 1 : 0
}

// Возврат хода и подсказка — это уже не своя игра; партия, брошенная до второго хода, — не партия.
export function isRated({ usedUndo, usedHint, playerMoves }) {
  return !usedUndo && !usedHint && playerMoves >= 2
}
```

```js
// src/practice/chess/clock.js
// Шахматные часы. Остаток считается от отметки времени, а не тиками: таймер в фоновой вкладке браузер
// замедляет, и часы на тиках отставали бы. Объект неизменяемый — машина партии хранит снимки по ходам
// и возвращает их при откате.
export function newClock(minutes) {
  if (!minutes) return null
  const ms = minutes * 60000
  return { w: ms, b: ms, running: null, since: null }
}

export const fromSnapshot = ({ w, b }) => ({ w, b, running: null, since: null })

export function remaining(clock, side, now) {
  const spent = clock.running === side ? now - clock.since : 0
  return Math.max(0, clock[side] - spent)
}

export function snapshot(clock, now) {
  return clock ? { w: remaining(clock, 'w', now), b: remaining(clock, 'b', now) } : null
}

export function pause(clock, now) {
  return clock ? { ...snapshot(clock, now), running: null, since: null } : null
}

export function switchTo(clock, side, now) {
  return clock ? { ...snapshot(clock, now), running: side, since: now } : null
}

export function flagged(clock, now) {
  if (!clock || !clock.running) return null
  return remaining(clock, clock.running, now) <= 0 ? clock.running : null
}
```

```js
import { createGame, kingSquare, capturedBy, materialLead, canMate } from './game.js'

const play = (g, ...sans) => { for (const s of sans) expect(g.move(sanToMove(g, s))).not.toBe(null) }
// Тесту удобнее SAN, а игра принимает клетки — находим ход перебором допустимых.
function sanToMove(g, san) {
  for (const row of g.board()) for (const cell of row) {
    if (!cell || cell.color !== g.turn()) continue
    for (const m of g.movesFrom(cell.square)) {
      for (const promotion of m.promotion ? ['q', 'r', 'b', 'n'] : [undefined]) {
        const probe = createGame(g.pgn())
        const made = probe.move({ from: cell.square, to: m.to, promotion })
        if (made && made.san === san) return { from: cell.square, to: m.to, promotion }
      }
    }
  }
  throw new Error(`нет хода ${san}`)
}

describe('партия', () => {

    const g = createGame()
    expect(g.turn()).toBe('w')

    expect(g.movesFrom('e7')).toEqual([])
  })
  it('недопустимый ход не делается и не бросает', () => {
    const g = createGame()
```

```js
// Партия поверх chess.js. Правила (ходы, шах, мат, повторения) — библиотечные; здесь только форма, в которой
// их видит остальной раздел. Про бота, часы и React модуль не знает — на нём же поедет онлайн.
import { Chess } from 'chess.js'

const VALUE = { p: 1, n: 3, b: 3, r: 5, q: 9 }

const record = (m) => ({
  san: m.san, from: m.from, to: m.to, color: m.color, piece: m.piece,
  captured: m.captured || null, promotion: m.promotion || null, flags: m.flags, before: m.before, after: m.after,
})

export function createGame(pgn = '') {
  const chess = new Chess()
  if (pgn) chess.loadPgn(pgn)
  return {
    fen: () => chess.fen(),
    turn: () => chess.turn(),
    pgn: () => chess.pgn(),
    ply: () => chess.history().length,
    board: () => chess.board(),
    inCheck: () => chess.inCheck(),
    moves: () => chess.history({ verbose: true }).map(record),
    // Превращение даёт четыре хода на одну клетку — доске нужна одна отметка, фигуру спросит выбор.
    movesFrom(square) {
      const seen = new Map()
      for (const m of chess.moves({ square, verbose: true })) {
        if (!seen.has(m.to)) seen.set(m.to, { to: m.to, capture: m.flags.includes('c') || m.flags.includes('e'), promotion: !!m.promotion })
      }
      return [...seen.values()]
    },
    // chess.js 1.x на недопустимый ход бросает; доске удобнее null.
    move({ from, to, promotion }) {
      try { return record(chess.move({ from, to, promotion })) } catch { return null }
    },
    undo() {
      const m = chess.undo()
      return m ? record(m) : null
    },
    outcome() {
      if (chess.isCheckmate()) return { result: chess.turn() === 'w' ? '0-1' : '1-0', reason: 'checkmate' }
      if (chess.isStalemate()) return { result: '1/2-1/2', reason: 'stalemate' }
      if (chess.isInsufficientMaterial()) return { result: '1/2-1/2', reason: 'insufficient' }
      if (chess.isThreefoldRepetition()) return { result: '1/2-1/2', reason: 'threefold' }
      if (chess.isDrawByFiftyMoves()) return { result: '1/2-1/2', reason: 'fifty' }
      return null
    },
  }
}

export function kingSquare(board, color) {
  for (const row of board) for (const cell of row) if (cell && cell.type === 'k' && cell.color === color) return cell.square
  return null
}

/** Что взяла каждая сторона, в порядке взятий. */
export function capturedBy(moves) {
  const out = { w: [], b: [] }
  for (const m of moves) if (m.captured) out[m.color].push(m.captured)
  return out
}

/** Перевес белых в пешках; превращение считается как прибавка разницы фигур. */
export function materialLead(moves) {
  let lead = 0
  for (const m of moves) {
    const sign = m.color === 'w' ? 1 : -1
    if (m.captured) lead += sign * VALUE[m.captured]
    if (m.promotion) lead += sign * (VALUE[m.promotion] - VALUE.p)
  }
  return lead
}

// Для ничьей по времени: упавший флаг — не поражение, если сопернику нечем ставить мат.
export function canMate(board, color) {
  const mine = board.flat().filter((c) => c && c.color === color && c.type !== 'k').map((c) => c.type)
  if (!mine.length) return false
  return !(mine.length === 1 && (mine[0] === 'n' || mine[0] === 'b'))
}
```

```js
// src/practice/chess/strength.js
// Рейтинг бота → настройки Stockfish. У движка шкала UCI_Elo начинается с 1320 (проверено запуском сборки
// 19 lite: min 1320 max 3190). Ниже — своя достройка: самый слабый уровень движка на малой глубине плюс доля
// случайных ходов. Она приблизительная: гарантируется только, что выше рейтинг — не слабее игра.
import { Chess } from 'chess.js'

export const MIN_BOT = 400
export const MAX_BOT = 2800
export const BOT_STEP = 100
export const NATIVE_FLOOR = 1320

export function clampBot(r) {
  const n = Number(r)
  if (!Number.isFinite(n)) return 800
  return Math.min(MAX_BOT, Math.max(MIN_BOT, Math.round(n / BOT_STEP) * BOT_STEP))
}

// Все три опции шлются на каждый поиск: подсказка и бот ходят одним движком, и забытая опция
// прошлого поиска сделала бы бота гроссмейстером на один ход.
export const FULL_STRENGTH = {
  options: { UCI_LimitStrength: 'false', UCI_Elo: NATIVE_FLOOR, 'Skill Level': 20 },
  go: { movetime: 500 },
  blunder: 0,
}

export function botSettings(rating) {
  const r = clampBot(rating)
  if (r >= NATIVE_FLOOR) {
    const share = (r - NATIVE_FLOOR) / (MAX_BOT - NATIVE_FLOOR)
    return {
      options: { UCI_LimitStrength: 'true', UCI_Elo: r, 'Skill Level': 20 },
      go: { movetime: Math.round(300 + share * 700) },
      blunder: 0,
    }
  }
  return {
    options: { UCI_LimitStrength: 'false', UCI_Elo: NATIVE_FLOOR, 'Skill Level': 0 },
    go: { depth: 1 + Math.floor((r - MIN_BOT) / 300) },
    blunder: 0.5 - (r - MIN_BOT) * 0.0005,
  }
}

const LEVELS = [[800, 'beginner'], [1200, 'novice'], [1600, 'club'], [2000, 'strongClub'], [2200, 'expert'], [2500, 'master']]
export function levelKey(rating) {
  const r = clampBot(rating)
  for (const [below, key] of LEVELS) if (r < below) return key
  return 'grandmaster'
}

/**
 * Случайный допустимый ход. careful — не ставить своего ферзя под бой одним ходом, если есть другие ходы:
 * без этого бот на 800 зевал бы ферзя чаще живого новичка. Генератор приходит аргументом — ради теста.
 */
export function weakMove(fen, rng, careful) {
  const chess = new Chess(fen)
  let moves = chess.moves({ verbose: true })
  if (!moves.length) return null
  if (careful) {
    const safe = moves.filter((m) => {
      chess.move({ from: m.from, to: m.to, promotion: m.promotion })
      const hangs = chess.moves({ verbose: true }).some((reply) => reply.captured === 'q')
      chess.undo()
      return !hangs
    })
    if (safe.length) moves = safe
  }
  const m = moves[Math.min(moves.length - 1, Math.floor(rng() * moves.length))]
  return m.promotion ? { from: m.from, to: m.to, promotion: m.promotion } : { from: m.from, to: m.to }
}
```

```js
// src/practice/chess/engineClient.js
// Клиент Stockfish. Движок — отдельная программа (GPL) в Web Worker: сайт его не импортирует, а переписывается
// с ним строками протокола UCI. Поиск один за раз: движок однопоточный, и вторая команда go поверх первой
// спутала бы, чей пришёл bestmove. Тот же клиент возьмёт тренер (часть 2) — search уже отдаёт оценку.
export const ENGINE_URL = '/chess/engine/stockfish-19-lite-single.js'
const READY_TIMEOUT = 20000
const CANCELLED = { move: null, score: null, cancelled: true }

export function parseBestMove(line) {
  const m = /^bestmove ([a-h][1-8])([a-h][1-8])([qrbn])?/.exec(line)
  if (!m) return null
  return m[3] ? { from: m[1], to: m[2], promotion: m[3] } : { from: m[1], to: m[2] }
}

export function parseScore(line) {
  const m = /^info .*\bscore (cp|mate) (-?\d+)/.exec(line)
  if (!m) return null
  return m[1] === 'cp' ? { cp: Number(m[2]) } : { mate: Number(m[2]) }
}

export function createEngine({ makeWorker = () => new Worker(ENGINE_URL), readyTimeout = READY_TIMEOUT, onCrash = () => {} } = {}) {
  let worker = null
  let dead = false
  let waiters = [] // [{ test, resolve }] — кто ждёт служебную строку (uciok, readyok)
  let search = null // { resolve, score, cancelled } — идущий поиск
  let queue = Promise.resolve()
  let rejectReady = () => {}

  const send = (cmd) => { try { worker?.postMessage(cmd) } catch { /* воркер уже погашен */ } }
  const waitFor = (test) => new Promise((resolve) => waiters.push({ test, resolve }))

  function fail(err) {
    if (dead) return
    dead = true
    waiters = []
    if (search) { search.resolve(CANCELLED); search = null }
    rejectReady(err)
    onCrash(err)
  }

  function onLine(line) {
    if (typeof line !== 'string') return
    waiters = waiters.filter((w) => {
      if (!w.test(line)) return true
      w.resolve()
      return false
    })
    if (!search) return
    if (line.startsWith('info ')) {
      const score = parseScore(line)
      if (score) search.score = score
    } else if (line.startsWith('bestmove')) {
      const done = search
      search = null
      done.resolve(done.cancelled ? CANCELLED : { move: parseBestMove(line), score: done.score, cancelled: false })
    }
  }

  const ready = new Promise((resolve, reject) => {
    rejectReady = reject
    try { worker = makeWorker() } catch (err) { dead = true; reject(err); onCrash(err); return }
    worker.onmessage = (ev) => onLine(ev.data)
    worker.onerror = (ev) => fail(new Error(ev?.message || 'engine error'))
    const timer = setTimeout(() => fail(new Error('engine timeout')), readyTimeout)
    waitFor((l) => l === 'uciok')
      .then(() => { const up = waitFor((l) => l === 'readyok'); send('isready'); return up })
      .then(() => { clearTimeout(timer); resolve() })
    send('uci')
  })
  // Отказ ready читает экран; без этого обработчика он всплыл бы необработанным отклонением.
  ready.catch(() => {})

  function run(fen, { options = {}, go = {} } = {}) {
    return new Promise((resolve) => {
      if (dead) { resolve(CANCELLED); return }
      search = { resolve, score: null, cancelled: false }
      for (const [name, value] of Object.entries(options)) send(`setoption name ${name} value ${value}`)
      send(`position fen ${fen}`)
      send(go.depth ? `go depth ${go.depth}` : `go movetime ${go.movetime || 500}`)
    })
  }

  return {
    ready,
    search(fen, settings) {
      const result = queue.then(() => ready).then(() => run(fen, settings), () => CANCELLED)
      queue = result
      return result
    },
    stop() {
      if (search && !search.cancelled) { search.cancelled = true; send('stop') }
    },
    destroy() {
      if (search) { search.resolve(CANCELLED); search = null }
      send('quit')
      dead = true
      try { worker?.terminate() } catch { /* уже погашен */ }
    },
  }
}
```

```js
// src/practice/chess/opponent.js
// Соперник — тот, у кого партия спрашивает ход. Сейчас это бот; онлайн-соперник (часть 4) реализует те же
// requestMove и cancel поверх сети, и машина партии не изменится.
import { botSettings, weakMove, MIN_BOT } from './strength.js'

const defaultSleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms))

export function createBotOpponent({ engine, rating, rng = Math.random, minDelay = 400, sleep = defaultSleep }) {
  // Номер запроса: ответ на отменённый (возврат хода, новая партия) не должен попасть на доску.
  let token = 0
  return {
    async requestMove(fen) {
      const mine = ++token
      const settings = botSettings(rating)
      // Мгновенный ответ выглядит как сбой, а звук хода сливается со звуком хода ученика.
      const pause = sleep(minDelay)
      let move
      if (rng() < settings.blunder) {
        move = weakMove(fen, rng, rating > MIN_BOT)
      } else {
        const found = await engine.search(fen, settings)
        if (found.cancelled) return null
        move = found.move
      }
      await pause
      return mine === token ? move : null
    },
    cancel() {
      token++
      engine.stop()
    },
  }
}
```

```js
let ctx = null
let master = null
let noiseBuf = null
let enabled = null

function readPrefs() {
  try { return JSON.parse(window.localStorage.getItem(CHESS_PREFS_KEY) || '{}') || {} } catch { return {} }

export function isChessSoundOn() {
  // По умолчанию включён: выключенный звук, о котором никто не знает, всё равно что его нет.
  if (enabled === null) enabled = typeof window === 'undefined' ? true : readPrefs().sound !== false
  return enabled
}

export function setChessSoundOn(on) {
  enabled = !!on
  try { window.localStorage.setItem(CHESS_PREFS_KEY, JSON.stringify({ ...readPrefs(), sound: enabled })) } catch { /* приватное окно */ }
  if (enabled) wakeChessSound()
}

export function wakeChessSound() {
  try {
    const AC = typeof window !== 'undefined' && (window.AudioContext || window.webkitAudioContext)
    if (!AC) return null
    if (!ctx) {
      ctx = new AC()
      master = ctx.createGain()
      master.gain.value = MASTER
      master.connect(ctx.destination)
    }
    if (ctx.state === 'suspended') ctx.resume().catch(() => {})
    return ctx
  } catch {
    return null
  }
}

export function sleepChessSound() {
  if (ctx && ctx.state === 'running') ctx.suspend().catch(() => {})
}

function noise() {
  if (!noiseBuf) {
    noiseBuf = ctx.createBuffer(1, Math.floor(ctx.sampleRate * 0.2), ctx.sampleRate)
    const data = noiseBuf.getChannelData(0)
    for (let i = 0; i < data.length; i++) data[i] = Math.random() * 2 - 1
  }
  return noiseBuf
}

// Стук дерева: короткий шум через фильтр плюс низкий удар.
function knock(at, peak, cutoff) {
  const t = ctx.currentTime + at
  const src = ctx.createBufferSource()
  src.buffer = noise()
  const filter = ctx.createBiquadFilter()
  filter.type = 'lowpass'
  filter.frequency.value = cutoff
  const gain = ctx.createGain()
  gain.gain.setValueAtTime(peak, t)
  gain.gain.exponentialRampToValueAtTime(0.001, t + 0.07)
  src.connect(filter).connect(gain).connect(master)
  src.start(t)
  src.stop(t + 0.08)
  tone(170, at, 0.07, peak * 0.5, 'sine')
}

function tone(freq, at, dur, peak, type = 'triangle') {
  const t = ctx.currentTime + at
  const osc = ctx.createOscillator()
  osc.type = type
  osc.frequency.value = freq
  const gain = ctx.createGain()
  gain.gain.setValueAtTime(0.0001, t)
  gain.gain.exponentialRampToValueAtTime(peak, t + 0.008)
  gain.gain.exponentialRampToValueAtTime(0.001, t + dur)
  osc.connect(gain).connect(master)
  osc.start(t)
  osc.stop(t + dur + 0.02)
}

const melody = (notes, step, dur, peak) => notes.forEach((f, i) => tone(f, i * step, dur, peak))

const CUES = {
  move: () => knock(0, 0.5, 1800),
  capture: () => { knock(0, 0.8, 2600); knock(0.05, 0.45, 1300) },
  castle: () => { knock(0, 0.5, 1800); knock(0.1, 0.5, 1800) },
  check: () => { knock(0, 0.5, 1800); tone(880, 0.03, 0.14, 0.2) },
  promote: () => { knock(0, 0.5, 1800); melody([523, 659, 784], 0.07, 0.16, 0.2) },
  win: () => melody([523, 659, 784, 1047], 0.12, 0.3, 0.25),
  lose: () => melody([392, 330, 262], 0.16, 0.36, 0.22),
  draw: () => melody([440, 440], 0.2, 0.26, 0.2),
  lowtime: () => { tone(1000, 0, 0.08, 0.18, 'square'); tone(1000, 0.16, 0.08, 0.18, 'square') },
  illegal: () => tone(140, 0, 0.09, 0.12, 'sine'),
}

export function playChessCue(name) {
  if (!isChessSoundOn() || !CUES[name]) return
  if (!wakeChessSound() || ctx.state !== 'running') return
  try { CUES[name]() } catch { /* звук — не причина ронять партию */ }
}
```

```js
    time: TIME_OPTIONS.includes(r.time) ? r.time : 0, result: r.result, reason: text(r.reason, 24) || 'checkmate', pgn,
    rated: r.rated === true, ratingBefore: int(r.ratingBefore, FLOOR, 4000, START_RATING), ratingAfter: int(r.ratingAfter, FLOOR, 4000, START_RATING),
  }
}

function parseCurrent(raw) {
  if (typeof raw !== 'string') return null
  try { return normCurrent(JSON.parse(raw)) } catch { return null }
}

/** Текущая партия объектом — или null, если её нет. */
export const currentOf = (state) => parseCurrent(state?.current)

export function normalize(raw) {
  const out = emptyState()
  if (!isObj(raw)) return out
  out.rating = int(raw.rating, FLOOR, 4000, START_RATING)
  out.games = int(raw.games, 0, 1e6)
  const current = parseCurrent(raw.current)
  out.current = current ? JSON.stringify(current) : null
  if (Array.isArray(raw.history)) {
    const byId = new Map()
    for (const r of raw.history.map(normRecord)) if (r && !byId.has(r.id)) byId.set(r.id, r)
    out.history = [...byId.values()].sort((a, b) => b.endedAt - a.endedAt).slice(0, HISTORY_MAX)
  }
  return out
}

/** Сведение двух копий: сыгранное где угодно — в истории; рейтинг — у копии, где партий больше. */
export function mergeChess(server, local) {
  const a = normalize(server)
  const b = normalize(local)
  const lead = b.games > a.games ? b : a
  const history = normalize({ history: [...a.history, ...b.history] }).history
  const ca = parseCurrent(a.current)
  const cb = parseCurrent(b.current)
  let current = !ca ? cb : !cb ? ca : cb.movedAt > ca.movedAt ? cb : ca
  // Партия, доигранная на другом устройстве, не должна вернуться недоигранной.
  if (current && history.some((r) => r.id === current.id)) current = null
  return { rating: lead.rating, games: lead.games, current: current ? JSON.stringify(current) : null, history }
}

const store = createProgressStore({

  key: KEY,
  event: EVENT,
  empty: emptyState,
  normalize,
  merge: (server, local) => mergeChess(server, local),
})

export function readChess() {
  return store.read()
}

function update(fn) {
  const next = normalize(JSON.parse(JSON.stringify(store.read())))
  fn(next)
  store.write(normalize(next))
}

export function saveCurrent(current) {
  update((s) => { s.current = current ? JSON.stringify(current) : null })
}

export function recordGame(record) {
  update((s) => {
    s.history = [record, ...s.history.filter((r) => r.id !== record.id)]
    s.games += 1
    if (record.rated) s.rating = record.ratingAfter
    s.current = null
  })
```

```js
// src/practice/chess/session.js
// Машина партии: чей ход, думает ли бот, часы, возврат хода, подсказка, исход и рейтинг.
// Живёт вне React (экран подписан через useSyncExternalStore), как машина «Неправильных глаголов»:
// ответ бота приходит позже и не должен зависеть от того, какой рендер был последним.
// О движке, хранилище и звуке не знает — всё приходит аргументами; сетевой соперник (часть 4)
// встанет на место makeOpponent.
import { createGame, kingSquare, capturedBy, materialLead, canMate } from './game.js'
import { newClock, fromSnapshot, remaining, switchTo, pause, flagged, snapshot } from './clock.js'
import { nextRating, scoreFor, isRated } from './rating.js'
import { cueOf } from './cues.js'

const LOW_TIME = 10000
const other = (side) => (side === 'w' ? 'b' : 'w')
const winOf = (side) => (side === 'w' ? '1-0' : '0-1')
const makeId = () => `g${Date.now().toString(36)}${Math.random().toString(36).slice(2, 8)}`

export function createChessSession({
  makeOpponent, analyse, save, finish, getRating,
  now = () => performance.now(), wallClock = () => Date.now(), onCue = () => {}, newId = makeId,
}) {
  let game = null
  let meta = null // { id, startedAt, color, botRating, time }
  let opponent = null
  let clock = null
  let clockLog = [] // clockLog[n] — остатки после n полуходов; по нему возврат хода возвращает часы
  let flags = { usedUndo: false, usedHint: false }
  let status = 'idle'
  let thinking = false
  let hint = null
  let hintPending = false
  let outcome = null
  let ratingChange = null
  let engineError = false
  let lowTimeCued = false
  const listeners = new Set()

  function build() {
    if (!game) {
      return {
        status, fen: null, board: null, turn: null, playerColor: null, botRating: null, time: 0, moves: [], lastMove: null,
        check: null, thinking: false, hint: null, hintPending: false, rated: true, canUndo: false, clock: null,
        captured: { w: [], b: [] }, lead: 0, outcome, ratingChange, engineError,
      }
    }
    const moves = game.moves()
    const last = moves[moves.length - 1]
    const board = game.board()
    const turn = game.turn()
    const undoSteps = turn === meta.color ? 2 : 1
    return {
      status, fen: game.fen(), board, turn, playerColor: meta.color, botRating: meta.botRating, time: meta.time,
      moves: moves.map(({ san, from, to, color }) => ({ san, from, to, color })),
      lastMove: last ? { from: last.from, to: last.to } : null,
      check: game.inCheck() ? kingSquare(board, turn) : null,
      thinking, hint, hintPending, rated: !flags.usedUndo && !flags.usedHint,
      canUndo: status === 'playing' && moves.length >= undoSteps,
      clock: snapshot(clock, now()), captured: capturedBy(moves), lead: materialLead(moves),
      outcome, ratingChange, engineError,
    }
  }
  let snap = build()
  function emit() {
    snap = build()
    for (const fn of listeners) fn()
  }

  function persist() {
    save({
      ...meta, pgn: game.pgn(), movedAt: wallClock(), clock: snapshot(clock, now()), clockLog: clockLog.slice(),
      usedUndo: flags.usedUndo, usedHint: flags.usedHint,
    })
  }

  function end(result, reason, cue = true) {
    opponent?.cancel()
    thinking = false
    hint = null
    hintPending = false
    clock = pause(clock, now())
    status = 'over'
    outcome = { result, reason }
    const playerMoves = game.moves().filter((m) => m.color === meta.color).length
    const rated = isRated({ ...flags, playerMoves })
    const before = getRating()
    const after = rated ? nextRating(before, meta.botRating, scoreFor(result, meta.color)) : before
    ratingChange = { before, after, rated }
    // Партия без единого хода — не партия: в историю не идёт.
    if (game.ply() > 0) {
      finish({
        id: meta.id, endedAt: wallClock(), color: meta.color, botRating: meta.botRating, time: meta.time,
        result, reason, pgn: game.pgn(), rated, ratingBefore: before, ratingAfter: after,
      })
    }
    save(null)
    if (cue) onCue(cueOf(null, outcome, meta.color))
    emit()
  }

  function afterMove(move) {
    hint = null
    if (clock) {
      const t = now()
      clock = switchTo(clock, game.turn(), t)
      clockLog[game.ply()] = snapshot(clock, t)
    }
    const over = game.outcome()
    onCue(cueOf(move, over, meta.color))
    if (over) { end(over.result, over.reason, false); return }
    persist()
    emit()
    if (game.turn() !== meta.color) askBot()
  }

  function askBot() {
    const fen = game.fen()
    const asked = opponent
    thinking = true
    emit()
    asked.requestMove(fen).then((move) => {
      // Ответ на позицию, которой уже нет (возврат хода, новая партия), на доску не идёт.
      if (!move || status !== 'playing' || opponent !== asked || game.fen() !== fen) return
      thinking = false
      const made = game.move(move)
      if (made) afterMove(made)
      else emit()
    }, () => {})
  }

  function begin(nextMeta) {
    meta = nextMeta
    opponent = makeOpponent(meta.botRating)
    status = 'playing'
    thinking = false
    hint = null
    hintPending = false
    outcome = null
    ratingChange = null
    engineError = false
    lowTimeCued = false
  }

  return {
    subscribe(fn) {
      listeners.add(fn)
      return () => listeners.delete(fn)
    },
    getSnapshot: () => snap,

    start({ color, botRating, time }) {
      // Бросить рейтинговую партию ради новой — проиграть её: иначе рейтинг накручивается уходом из проигранных.
      if (status === 'playing') end(winOf(other(meta.color)), 'abandoned', false)
      game = createGame()
      flags = { usedUndo: false, usedHint: false }
      clock = newClock(time)
      clockLog = clock ? [snapshot(clock, now())] : []
      begin({ id: newId(), startedAt: wallClock(), color, botRating, time })
      persist()
      emit()
      if (color === 'b') askBot()
    },

    resume(saved) {
      let restored
      try { restored = createGame(saved.pgn) } catch { restored = null }
      if (!restored || restored.outcome()) { save(null); return false }
      game = restored
      flags = { usedUndo: !!saved.usedUndo, usedHint: !!saved.usedHint }
      const ply = game.ply()
      clock = saved.clock ? fromSnapshot(saved.clock) : null
      // Журнал другой длины (старая запись, сведение устройств) — заполняем текущими остатками.
      clockLog = !clock ? [] : saved.clockLog?.length === ply + 1 ? saved.clockLog.slice() : Array.from({ length: ply + 1 }, () => ({ ...saved.clock }))
      begin({ id: saved.id, startedAt: saved.startedAt, color: saved.color, botRating: saved.botRating, time: clock ? saved.time : 0 })
      // Перезагрузка часы останавливала; идут снова с сохранённого остатка.
      if (clock && ply > 0) clock = switchTo(clock, game.turn(), now())
      emit()
      if (game.turn() !== meta.color) askBot()
      return true
    },

    movesFrom(square) {
      if (status !== 'playing' || engineError || thinking || game.turn() !== meta.color) return []
      return game.movesFrom(square)
    },

    playerMove(from, to, promotion) {
      if (status !== 'playing' || engineError || thinking || game.turn() !== meta.color) return false
      const made = game.move({ from, to, promotion })
      if (!made) { onCue('illegal'); return false }
      afterMove(made)
      return true
    },

    undo() {
      if (status !== 'playing') return false
      // Бот уже ответил — снимаем его ход и свой; ещё думает — только свой.
      const steps = game.turn() === meta.color ? 2 : 1
      if (game.ply() < steps) return false
      opponent.cancel()
      thinking = false
      hint = null
      for (let i = 0; i < steps; i++) game.undo()
      flags.usedUndo = true
      if (clock) {
        const ply = game.ply()
        clock = fromSnapshot(clockLog[ply] || snapshot(clock, now()))
        clockLog.length = ply + 1
        if (ply > 0) clock = switchTo(clock, game.turn(), now())
      }
      persist()
      emit()
      return true
    },

    hint() {
      if (status !== 'playing' || engineError || thinking || hintPending || game.turn() !== meta.color) return
      const fen = game.fen()
      flags.usedHint = true
      hintPending = true
      persist()
      emit()
      const settle = (move) => {
        hintPending = false
        if (move && status === 'playing' && game.fen() === fen) hint = { from: move.from, to: move.to }
        emit()
      }
      analyse(fen).then(settle, () => settle(null))
    },

    resign() {
      if (status === 'playing') end(winOf(other(meta.color)), 'resignation')
    },

    // Экран зовёт раз в 100 мс: обновить часы на экране, поймать флаг и «мало времени».
    tick() {
      if (status !== 'playing' || !clock || engineError) return
      const t = now()
      const side = flagged(clock, t)
      if (side) {
        // Флаг — не поражение, если сопернику нечем ставить мат.
        end(canMate(game.board(), other(side)) ? winOf(other(side)) : '1/2-1/2', 'timeout')
        return
      }
      if (!lowTimeCued && clock.running === meta.color && remaining(clock, meta.color, t) <= LOW_TIME) {
        lowTimeCued = true
        onCue('lowtime')
      }
      emit()
    },

    engineFailed() {
      if (status !== 'playing') return
      engineError = true
      thinking = false
      hintPending = false
      clock = pause(clock, now())
      emit()
    },

    retry() {
      if (status !== 'playing' || !engineError) return
      engineError = false
      opponent = makeOpponent(meta.botRating)
      if (clock && game.ply() > 0) clock = switchTo(clock, game.turn(), now())
      emit()
      if (game.turn() !== meta.color) askBot()
    },

    destroy() {
      opponent?.cancel()
      listeners.clear()
    },
  }
}
```

```jsx
'use client'

// Доска. Правил не знает: позицию, допустимые ходы выбранной фигуры и колбэк хода получает пропсами —
// поэтому та же доска показывает и живую партию, и разбор сыгранной (interactive = false).
// Ход — перетаскиванием или двумя тапами; возможные ходы — точка на пустой клетке и кольцо на клетке
// со взятием (приём Lichess, код свой).
import { useRef, useState } from 'react'
import { Piece } from './pieces.jsx'

const FILES = 'abcdefgh'
const PROMOTIONS = ['q', 'r', 'b', 'n']
// Сдвиг указателя, с которого нажатие считается перетаскиванием, а не тапом.
const DRAG_START = 6

export function squaresFor(orientation) {
  const files = orientation === 'w' ? [...FILES] : [...FILES].reverse()
  const ranks = orientation === 'w' ? [8, 7, 6, 5, 4, 3, 2, 1] : [1, 2, 3, 4, 5, 6, 7, 8]
  return ranks.flatMap((r) => files.map((f) => f + r))
}

const pieceAt = (board, square) => board[8 - Number(square[1])][FILES.indexOf(square[0])]

export default function Board({ fen, board, orientation, playerColor, interactive, movesFrom, onMove, lastMove, check, hint, animateLast, t }) {
  const rootRef = useRef(null)
  // Выбор привязан к позиции: сходил бот или вернули ход — старый выбор недействителен сам, без эффекта.
  const [picked, setPicked] = useState(null) // { fen, square }
  const [promo, setPromo] = useState(null) // { fen, from, to }
  const [drag, setDrag] = useState(null) // { from, startX, startY, x, y, moved }
  const squares = squaresFor(orientation)
  const selected = picked && picked.fen === fen ? picked.square : null
  const asking = promo && promo.fen === fen ? promo : null
  const targets = selected ? movesFrom(selected) : []
  const colors = { w: t('white').toLowerCase(), b: t('black').toLowerCase() }

  const tryMove = (from, to) => {
    const move = movesFrom(from).find((m) => m.to === to)
    setPicked(null)
    if (!move) return false
    if (move.promotion) setPromo({ fen, from, to })
    else onMove(from, to, undefined)
    return true
  }

  const press = (square) => {
    if (!interactive || asking) return false
    if (selected && selected !== square && targets.some((m) => m.to === square)) { tryMove(selected, square); return false }
    const piece = pieceAt(board, square)
    if (piece && piece.color === playerColor) { setPicked({ fen, square }); return true }
    setPicked(null)
    return false
  }

  const onPointerDown = (square) => (e) => {
    if (e.button > 0) return
    if (!press(square)) return
    // Захват на доске: дальше движение и отпускание приходят ей, где бы ни был указатель.
    try { rootRef.current?.setPointerCapture(e.pointerId) } catch { /* указатель уже отпущен */ }
    setDrag({ from: square, startX: e.clientX, startY: e.clientY, x: e.clientX, y: e.clientY, moved: false })
  }

  const onPointerMove = (e) => {
    if (!drag) return
    const moved = drag.moved || Math.hypot(e.clientX - drag.startX, e.clientY - drag.startY) > DRAG_START
    setDrag({ ...drag, x: e.clientX, y: e.clientY, moved })
  }

  const squareAt = (x, y) => {
    const rect = rootRef.current.getBoundingClientRect()
    const col = Math.floor(((x - rect.left) / rect.width) * 8)
    const row = Math.floor(((y - rect.top) / rect.height) * 8)
    return col >= 0 && col < 8 && row >= 0 && row < 8 ? squares[row * 8 + col] : null
  }

  const onPointerUp = (e) => {
    if (!drag) return
    const { from, moved } = drag
    setDrag(null)
    if (!moved) return
    const to = squareAt(e.clientX, e.clientY)
    // Бросили мимо допустимой клетки — фигура возвращается и остаётся выбранной.
    if (to && to !== from && !tryMove(from, to)) setPicked({ fen, square: from })
  }

  const onKeyDown = (square) => (e) => {
    if (e.key !== 'Enter' && e.key !== ' ') return
    e.preventDefault()
    press(square)
  }
```

```jsx
  const session = sessionRef.current
  const snap = useSyncExternalStore(session.subscribe, session.getSnapshot, session.getSnapshot)

  // Движок: поднять при входе, погасить при уходе; «Retry» пересоздаёт его (engineRun).
  useEffect(() => {
    let alive = true
    const { engine, fake } = makeEngine(() => { if (alive) { setEngineState('failed'); session.engineFailed() } })
    engineRef.current = engine
    fakeRef.current = fake
    setEngineState('loading')
    engine.ready.then(() => { if (alive) setEngineState('ready') }, () => {})
    if (engineRun === 0) {
      // Недоигранная партия возвращается сама; ждать движок не нужно — его поиск встанет в очередь за ready.
      const saved = currentOf(readChess())
      if (!saved || !session.resume(saved)) setDialog('new')
    } else {
      session.retry()
    }
    return () => {
      alive = false
      engine.destroy()
    }
  }, [engineRun, session])

  useEffect(() => () => { session.destroy(); sleepChessSound() }, [session])

  // Часы на экране и флаг — раз в 100 мс, только пока идёт партия с часами.
  const ticking = snap.status === 'playing' && !!snap.clock
  useEffect(() => {
    if (!ticking) return undefined
    const id = setInterval(() => session.tick(), 100)
    return () => clearInterval(id)
  }, [ticking, session])

  const pickLang = (next) => { setLang(next); writePrefs({ lang: next }) }
  const toggleSound = () => { const next = !soundOn; setChessSoundOn(next); setSoundOn(next) }

  const play = ({ color, botRating, time }) => {
    wakeChessSound()
    writePrefs({ color, botRating, time })
    const side = color === 'random' ? (Math.random() < 0.5 ? 'w' : 'b') : color
    session.start({ color: side, botRating, time })
    setFlipped(false)
    setDialog(null)
    setTab('play')
  }
  const askNew = () => {
    if (snap.status !== 'playing' || snap.moves.length === 0) { setDialog('new'); return }
    setDialog('confirmNew')
  }
  const move = (from, to, promotion) => { wakeChessSound(); session.playerMove(from, to, promotion) }

  const orientation = snap.playerColor ? (flipped ? (snap.playerColor === 'w' ? 'b' : 'w') : snap.playerColor) : 'w'
  const top = orientation === 'w' ? 'b' : 'w'
  const names = (side) => side === snap.playerColor
    ? { name: userName?.trim() || t('you'), sub: String(progress.rating) }
    : { name: t('bot'), sub: snap.thinking ? t('thinking') : String(snap.botRating ?? '') }
  const lastByBot = snap.moves.length > 0 && snap.moves[snap.moves.length - 1].color !== snap.playerColor
  const showResult = snap.status === 'over' && resultSeen !== snap.fen && !dialog
  // Бросить партию после второго своего хода без возвратов и подсказок — проиграть её (см. session.start).
  const rated = snap.rated && snap.moves.filter((m) => m.color === snap.playerColor).length >= 2
  const prefs = readPrefs()

  return (
    <div className="ch" data-status={snap.status} data-turn={snap.turn || ''} data-engine={engineState} lang={lang}>
      <PieceSprite />
      <header className="ch-head">
        <h1 className="ch-head__title">{t('title')}</h1>
        <div className="ch-tabs" role="tablist">
          {[['play', 'tabPlay'], ['games', 'tabGames']].map(([key, label]) => (
            <button key={key} type="button" role="tab" aria-selected={tab === key} data-tab={key}
              className={`ch-tabs__item ${tab === key ? 'is-on' : ''}`} onClick={() => { setTab(key); setReview(null) }}>{t(label)}</button>
          ))}
        </div>
        <div className="ch-head__tools">
          <button type="button" className="ch-tool" data-act="sound" aria-pressed={soundOn} onClick={toggleSound}>{soundOn ? t('soundOn') : t('soundOff')}</button>
          <div className="ch-seg ch-seg--small" role="radiogroup" aria-label="Language">
            {CHESS_LANGS.map((code) => (
              <button key={code} type="button" role="radio" aria-checked={lang === code} data-lang={code}
                className={`ch-seg__item ${lang === code ? 'is-on' : ''}`} onClick={() => pickLang(code)}>{code.toUpperCase()}</button>
            ))}
          </div>
        </div>
      </header>

      {tab === 'games' && (
        <HistoryView t={t} history={progress.history} open={review} onOpen={setReview} onClose={() => setReview(null)} />
      )}

      {tab === 'play' && snap.board && (
        <div className="ch-game">
          <div className="ch-stage">
            <Board fen={snap.fen} board={snap.board} orientation={orientation} playerColor={snap.playerColor}
              interactive={snap.status === 'playing'} movesFrom={session.movesFrom} onMove={move}
              lastMove={snap.lastMove} check={snap.check} hint={snap.hint} animateLast={lastByBot} t={t} />
          </div>
          <aside className="ch-side">
            <PlayerBar {...names(top)} side={top} snap={snap} />
            {!snap.rated && snap.status === 'playing' && <p className="ch-unrated" data-unrated>{t('unrated')}</p>}
            {snap.engineError && (
              <p className="ch-alert" role="alert">{t('engineStopped')} <button type="button" className="ch-link" data-act="retry" onClick={() => setEngineRun((n) => n + 1)}>{t('retry')}</button></p>
            )}
            <MoveList moves={snap.moves} />
            <Actions t={t} snap={snap} onUndo={() => session.undo()} onHint={() => session.hint()} onFlip={() => setFlipped((f) => !f)}
              onResign={() => setDialog('resign')} onNew={askNew} />
            <PlayerBar {...names(orientation)} side={orientation} snap={snap} />
          </aside>
        </div>
      )}

      {dialog === 'new' && (
        <NewGameDialog t={t} rating={progress.rating} defaults={prefs} engine={engineState} onPlay={play}
          onRetry={() => setEngineRun((n) => n + 1)} onClose={snap.board ? () => setDialog(null) : null} />
      )}
      {(dialog === 'resign' || dialog === 'confirmNew') && (
        <div className="ch-dialog" data-dialog="confirm" role="dialog" aria-modal="true">
          <div className="ch-dialog__card">
            <p className="ch-dialog__text">{dialog === 'resign' ? t('confirmResign') : rated ? t('confirmNewRated') : t('confirmNew')}</p>
            <div className="ch-dialog__actions">
              <button type="button" className="ch-btn ch-btn--ghost" onClick={() => setDialog(null)}>{t('cancel')}</button>
              <button type="button" className="ch-btn ch-btn--main" data-act="confirm"
                onClick={() => { if (dialog === 'resign') { session.resign(); setDialog(null) } else setDialog('new') }}>
                {dialog === 'resign' ? t('resign') : t('newGame')}
              </button>
            </div>
          </div>
        </div>
      )}
      {showResult && (
        <ResultDialog t={t} snap={snap}
          onRematch={() => play({ color: snap.playerColor === 'w' ? 'b' : 'w', botRating: snap.botRating, time: snap.time })}
          onNew={() => setDialog('new')}
          onReview={() => setResultSeen(snap.fen)} />
      )}
    </div>
  )
}
```

```jsx
// src/screens/chess/HistoryView.jsx
'use client'

// Сыгранные партии и их разбор по ходам. Та же доска, что в игре, но без ввода. На этом экране позже
// появится тренер (часть 2): позиции каждого хода уже здесь.
import { useMemo, useState } from 'react'
import Board from './Board.jsx'
import { MoveList } from './GamePanel.jsx'
import { createGame, boardOf } from '../../practice/chess/game.js'
import { scoreFor } from '../../practice/chess/rating.js'

const HEADS = { 1: 'won', 0: 'lost', 0.5: 'draw' }
const START = 'rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1'
const none = () => []

function movesOf(pgn) {
  try { return createGame(pgn).moves() } catch { return null }
}

function Review({ t, record, onClose }) {
  const moves = useMemo(() => movesOf(record.pgn), [record.pgn])
  const [ply, setPly] = useState(moves ? moves.length : 0)
  if (!moves) {
    return (
      <div className="ch-review">
        <button type="button" className="ch-btn" onClick={onClose}>{t('back')}</button>
        <p className="ch-alert">{t('badGame')}</p>
      </div>
    )
  }
  const go = (n) => setPly(Math.max(0, Math.min(moves.length, n)))
  const fen = ply === 0 ? START : moves[ply - 1].after
  const last = ply === 0 ? null : { from: moves[ply - 1].from, to: moves[ply - 1].to }
  const onKeyDown = (e) => {
    if (e.key === 'ArrowLeft') go(ply - 1)
    else if (e.key === 'ArrowRight') go(ply + 1)
    else return
    e.preventDefault()
  }
  const steps = [['first', '⏮', 0], ['prev', '◀', ply - 1], ['next', '▶', ply + 1], ['last', '⏭', moves.length]]
  return (
    // tabIndex: стрелки листают партию, когда фокус где угодно внутри разбора
    <div className="ch-review ch-game" tabIndex={-1} onKeyDown={onKeyDown}>
      <div className="ch-stage">
        <Board fen={fen} board={boardOf(fen)} orientation={record.color} playerColor={record.color} interactive={false}
          movesFrom={none} onMove={none} lastMove={last} check={null} hint={null} animateLast={false} t={t} />
      </div>
      <aside className="ch-side">
        <button type="button" className="ch-btn" onClick={onClose}>{t('back')}</button>
        <MoveList moves={moves} current={ply} onPick={go} />
        <div className="ch-steps">
          {steps.map(([key, glyph, to]) => (
            <button key={key} type="button" className="ch-btn" data-step={key} aria-label={t(key)} onClick={() => go(to)}>{glyph}</button>
          ))}
        </div>
      </aside>
    </div>
  )
}

export default function HistoryView({ t, history, open, onOpen, onClose }) {
  if (open) return <Review key={open.id} t={t} record={open} onClose={onClose} />
  if (!history.length) return <p className="ch-empty">{t('gamesEmpty')}</p>
  return (
    <ul className="ch-games">
      {history.map((r) => {
        const score = scoreFor(r.result, r.color)
        const count = movesOf(r.pgn)?.length ?? 0
        return (
          <li key={r.id}>
            <button type="button" className={`ch-games__row ch-games__row--${HEADS[score]}`} data-game={r.id} onClick={() => onOpen(r)}>
              <b>{t(HEADS[score])}</b>
              <span>{t(`reason.${r.reason}`)}</span>
              <span>{t(r.color === 'w' ? 'white' : 'black')} · {t('vs', { rating: r.botRating })}</span>
              <span>{t('movesCount', { n: count })}</span>
              <span>{r.rated ? `${r.ratingBefore} → ${r.ratingAfter}` : '—'}</span>
              <time dateTime={new Date(r.endedAt).toISOString()}>{new Date(r.endedAt).toLocaleDateString()}</time>
            </button>
          </li>
        )
      })}
    </ul>
  )
}
```
